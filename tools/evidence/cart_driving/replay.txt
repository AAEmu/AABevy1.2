# Bevy replay of the 1.2 cart driving cases (S6/S7).
# The native capture is tools/evidence/cart_driving/native.json.
# The cart is slave template 60 (Farm Wagon); it spawns 5 m in front of the character,
# so with the character at 10512 15585 facing 0 the cart is at (10512, 15590).
lua if X2Mate:IsPlayerMounting() then X2Mate:UnMountMate() end
wait 1
lua X2SiegeWeapon:DiscardSiegeWeapon()
waitfor 10 !riding
wait 2
gm move 10512 15585 230.5
wait 3
face 0
gm slave spawn 60
wait 5
units 60
gm move 10512 15588 230.5
wait 3
face 0
units 60
interact name:Wagon
waitfor 10 riding
wait 2

# (a) still 5 s: the send pump must resend the same body
echo case=a_still
wait 5

# (b) forward 5 s + coast
echo case=b_forward
hold W 5
wait 2.5

# (c) hard left 3 s (W starts 0.1 s before A, as in the capture)
echo case=c_left
hold W 0.1
hold WA 3.0
hold W 0.3
wait 2.3

# (c) hard right 3 s
echo case=c_right
hold W 0.1
hold WD 3.0
hold W 0.3
wait 2.3

# (d) reverse 3 s
echo case=d_reverse
hold S 3.0
wait 2.5

# (f) leave while moving: W held 1.5 s, then DiscardSiegeWeapon (0x032)
echo case=f_leave
hold W 1.5
lua X2SiegeWeapon:DiscardSiegeWeapon()
wait 1
expect !riding
wait 2

quit now
