# Replay of the behind, right and left approaches in the Bevy client (slot
# wp1b). The horse is summoned at the standard spot; the player walks around it
# and presses the mount action, as the 1.2 probe did.
gm move 10512 15585 232
wait 3
face 0
use 8159
wait 10
# expect 3: behind the horse, which faces north
lua X2Mate:MountMate()
waitfor 30 riding
wait 3
lua X2Mate:UnMountMate()
waitfor 20 !riding
wait 2
# east of the horse, facing west
walk 10516 15588
face 90
# expect 2: on the horse's right (east)
lua X2Mate:MountMate()
waitfor 30 riding
wait 3
lua X2Mate:UnMountMate()
waitfor 20 !riding
wait 2
# west of the horse, facing east
walk 10508 15588
face 270
# expect 1: on the horse's left (west)
lua X2Mate:MountMate()
waitfor 30 riding
wait 3
lua X2Mate:UnMountMate()
waitfor 20 !riding
wait 2
quit now
