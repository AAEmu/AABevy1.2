1: //! `.caf` animation clips: per-bone rotation / position tracks.
2: //!
3: //! A `.caf` is a CryEngine chunk file (`0xFFFF0001`, container 0x744 or 0x745) holding
4: //!
5: //! * one **Timing** chunk (`0xCCCC000E`, v0x918): `f32 secs_per_tick, i32 ticks_per_frame,
6: //!   char name[32], i32 start_frame, i32 end_frame` (always 1/4800 s per tick, 160 ticks per
7: //!   frame = 30 fps);
8: //! * one **Controller** chunk (`0xCCCC000D`) per animated bone; the controller id is the CRC-32 of
9: //!   the case-sensitive bone name ([`aa_cry::skeleton::controller_id`]);
10: //! * `0xAAFC0002` / `0xAAFC0003` (motion parameters, foot plants): ignored.
11: //!
12: //! Controller chunk **v0x829** (25,045 of the 25,202 `.caf` in the 1.2 pak, every player,
13: //! NPC and animal clip), after the 16-byte header copy:
14: //!
15: //! ```text
16: //! +0  u32 controller_id
17: //! +4  u16 num_rot_keys        +6  u16 num_pos_keys
18: //! +8  u8  rot_format          +9  u8  rot_time_format
19: //! +10 u8  pos_format          +11 u8  pos_keys_info (1 = own pos times, 0 = share rot times)
20: //! +12 u8  pos_time_format     +13 u8  tracks_aligned     (+14..16 padding)
21: //! +16 rot keys, rot times, [pos keys, pos times]
22: //! ```
23: //!
24: //! Rotation formats: 1 = `f32 x,y,z,w`; 5 = 48-bit "small tree" quaternion; 8 = 64-bit extended
25: //! small tree quaternion. A small-tree quaternion stores the three smallest components
26: //! quantised to `[-1/sqrt2, 1/sqrt2]` plus the 2-bit index of the largest one in the top bits
27: //! (48 bit: 3x15 bits, index at bits 46-47; 64 bit: 21+21+20 bits, index at bits 62-63, lowest
28: //! bits first, in component order x,y,z,w skipping the largest); the largest is rebuilt from
29: //! unit length. Position format 2 = `f32 x,y,z` (metres, relative to the parent bone). Time
30: //! formats: 0 = `f32`, 1 = `u16`, 2 = `u8`, all in **frames**.
31: //!
32: //! Values are the bone's *absolute local transform* (relative to its parent, replacing the
33: //! bind pose), not a delta. A bone without a position track keeps its bind position. The root
34: //! `Bip01` is the identity at the origin in idle clips (root motion in locomotion clips).
35: //!
36: //! Older chunk versions (v0x827, v0x828; 399 clips: ships, glider, sealion ...) are skipped
37: //! and counted in [`Clip::skipped_controllers`].

116:         let file = ChunkFile::parse(bytes)?;

121:         let mut start_frame = 0i32;
122:         let mut end_frame = 0i32;
123:         if let Some(t) = file.first(ty::TIMING) {

126:             let ticks_per_frame = r.i32()?;

128:             start_frame = r.i32()?;
129:             end_frame = r.i32()?;

131:             if spf > 1e-6 && spf.is_finite() {

169: fn decode_829(data: &[u8], fps: f32, frame_offset: f32) -> Result<Track> {
170:     let mut r = Reader::new(data);
171:     let controller = r.u32()?;
172:     let num_rot = r.u16()? as usize;
173:     let num_pos = r.u16()? as usize;
174:     let rot_format = r.u8()?;
175:     let rot_time_format = r.u8()?;
176:     let pos_format = r.u8()?;
177:     let pos_keys_info = r.u8()?;
178:     let pos_time_format = r.u8()?;
179:     let aligned = r.u8()? != 0;
180:     r.seek(16);
181: 
182:     let align = |r: &mut Reader<'_>| {
183:         if aligned {
184:             let p = (r.pos() + 3) & !3;
185:             r.seek(p);
186:         }
187:     };
188: 
189:     let mut track = Track { controller, ..Default::default() };
190: 
191:     // rotations
192:     if num_rot > 0 {
193:         let stride = match rot_format {
194:             1 => 16,
195:             5 => 6,
196:             8 => 8,
197:             f => return Err(Error::Format(format!("controller {controller:#x}: rotation format {f}"))),
198:         };
199:         let raw = r.bytes(num_rot * stride)?;
200:         track.rot = raw
201:             .chunks_exact(stride)
202:             .map(|k| match rot_format {
203:                 1 => Quat::from_xyzw(f(k, 0), f(k, 4), f(k, 8), f(k, 12)),
204:                 5 => small_tree_48(k),
205:                 _ => small_tree_64_ext(k),
206:             })
207:             .map(|q| if q.is_finite() && q.length_squared() > 0.5 { q.normalize() } else { Quat::IDENTITY })
208:             .collect();
209:         align(&mut r);
210:         track.rot_times = read_times(&mut r, num_rot, rot_time_format, fps, frame_offset)?;
211:         align(&mut r);
212:     }
213: 
214:     // positions
215:     if num_pos > 0 {
216:         if pos_format != 2 {
217:             return Err(Error::Format(format!("controller {controller:#x}: position format {pos_format}")));
218:         }
219:         let raw = r.bytes(num_pos * 12)?;
220:         track.pos = raw.chunks_exact(12).map(|k| Vec3::new(f(k, 0), f(k, 4), f(k, 8))).collect();
221:         align(&mut r);
222:         if pos_keys_info == 1 {
223:             track.pos_times = read_times(&mut r, num_pos, pos_time_format, fps, frame_offset)?;
224:         } else if num_pos == num_rot {
225:             track.pos_times = track.rot_times.clone();
226:         } else if num_pos == 1 {
227:             track.pos_times = vec![0.0];
228:         } else {

231:             )));
232:         }
233:     }
